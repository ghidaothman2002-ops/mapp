import {useEffect,useMemo,useRef,useState} from 'react';
import {geoMercator,geoPath} from 'd3-geo';
import {ArrowUpLeft,ArrowRight,Compass,Leaf,MapPin,Plus,Minus,RotateCcw,X,SlidersHorizontal,ChevronLeft,Mountain,Info,Sprout} from 'lucide-react';
import geography from './data/saudi-regions.geojson?raw';
import islandsData from './data/coastal-islands.geojson?raw';
import {regions,regionNames,sources} from './data/regions';
const geo=JSON.parse(geography);
const islands=JSON.parse(islandsData);
const image=(name)=>`${import.meta.env.BASE_URL}images/${name}.svg`;
function MapView({selected,onSelect}){
 const [view,setView]=useState({x:0,y:0,k:1});const [hover,setHover]=useState(null);const svg=useRef(null);const pointers=useRef(new Map());const gesture=useRef(null);const moved=useRef(false);
 const path=useMemo(()=>geoPath(geoMercator().fitExtent([[48,25],[732,565]],geo)),[]);
 function reset(){setView({x:0,y:0,k:1});}
 function zoom(factor){setView(v=>({...v,k:Math.max(1,Math.min(3,v.k*factor)),...(v.k*factor<=1?{x:0,y:0}:{})}));}
 const point=e=>{const r=svg.current.getBoundingClientRect();return {x:(e.clientX-r.left)*780/r.width,y:(e.clientY-r.top)*600/r.height};};
 function start(e){pointers.current.set(e.pointerId,point(e));moved.current=false;gesture.current={view,points:[...pointers.current.values()]};svg.current.setPointerCapture(e.pointerId);}
 function move(e){if(!pointers.current.has(e.pointerId))return;pointers.current.set(e.pointerId,point(e));const pts=[...pointers.current.values()],old=gesture.current.points,v=gesture.current.view;if(pts.length===2&&old.length===2){const dist=a=>Math.hypot(a[0].x-a[1].x,a[0].y-a[1].y);moved.current=true;setView({...v,k:Math.max(1,Math.min(3,v.k*dist(pts)/Math.max(1,dist(old))))});}else if(pts.length===1&&old.length===1){const dx=pts[0].x-old[0].x,dy=pts[0].y-old[0].y;if(Math.hypot(dx,dy)>5){moved.current=true;if(v.k>1)setView({...v,x:Math.max(-500,Math.min(500,v.x+dx)),y:Math.max(-350,Math.min(350,v.y+dy))});}}}
 function end(e){const initial=gesture.current?.points?.[0];if(!moved.current&&initial){const target=document.elementFromPoint(e.clientX,e.clientY)?.closest('[data-region]');if(target)onSelect(target.dataset.region);}pointers.current.delete(e.pointerId);gesture.current={view,points:[...pointers.current.values()]};}
 return <section className="map-section" aria-label="خريطة مناطق السعودية"><div className="map-top"><span><Compass size={17}/> أرض واحدة، حكايات كثيرة</span><span className="map-index">01 — 13</span></div><div className="map-canvas"><div className="map-watermark">المملكة العربية السعودية</div><svg ref={svg} viewBox="0 0 780 600" className="saudi-map" aria-label="اختر منطقة من الخريطة" onPointerDown={start} onPointerMove={move} onPointerUp={end} onPointerCancel={e=>{pointers.current.delete(e.pointerId);pointers.current.clear();gesture.current=null;}}>
 <g transform={`translate(${view.x+390} ${view.y+300}) scale(${view.k}) translate(-390 -300)`}><path d={path(islands)} className="coastal-islands" pointerEvents="none"><title>جزر ساحلية دون تعيين إداري في المصدر</title></path>{geo.features.map(f=><g key={f.id}><path d={path(f)} data-region={f.id} className={`region ${regions[f.id]?'available':''} ${selected===f.id?'selected':''}`} tabIndex="0" role="button" aria-label={`منطقة ${regionNames[f.id]}${regions[f.id]?'':'، المحتوى قريبًا'}`} aria-pressed={selected===f.id} onPointerEnter={()=>setHover(f.id)} onPointerLeave={()=>setHover(null)} onFocus={()=>setHover(f.id)} onBlur={()=>setHover(null)} onKeyDown={e=>{if(e.key==='Enter'||e.key===' '){e.preventDefault();onSelect(f.id);}}}><title>{regionNames[f.id]}</title></path>{regions[f.id]&&<g pointerEvents="none" transform={`translate(${path.centroid(f).join(' ')})`} className={`region-label ${selected===f.id?'active':''}`}><circle r="4"/><text y="-13" textAnchor="middle">{regionNames[f.id]}</text></g>}</g>)}</g></svg><div className="map-controls"><button onClick={()=>zoom(1.25)} aria-label="تكبير الخريطة" disabled={view.k===3}><Plus size={19}/></button><button onClick={()=>zoom(.8)} aria-label="تصغير الخريطة" disabled={view.k===1}><Minus size={19}/></button><span/><button onClick={reset} aria-label="إعادة ضبط الخريطة"><RotateCcw size={18}/></button></div><div className="north">N<span>↑</span></div><div className="hover-name" aria-live="polite">{hover?regionNames[hover]:'اختر منطقة لتبدأ الحكاية'}</div></div><div className="map-bottom"><span><i className="legend-dot"/> مناطق بانتظار اكتشافك</span><span><i className="legend-dot muted"/> محتوى قريبًا</span><small>كبّر واسحب لاستكشاف التفاصيل</small></div><div className="region-shortcuts" aria-label="اختيار المنطقة">{Object.entries(regions).map(([id,r])=><button className={selected===id?'active':''} key={id} onClick={()=>onSelect(id)}>{r.name}</button>)}</div></section>;
}
function Modal({item,onClose}){const ref=useRef(null);useEffect(()=>{ref.current.showModal();},[]);return <dialog ref={ref} onCancel={onClose} onClick={e=>{if(e.target===ref.current)onClose();}} className="detail-dialog"><button className="close" onClick={onClose} aria-label="إغلاق التفاصيل"><X/></button>{item.sources?<><h2>مصادر وشفافية المحتوى</h2><p>المشروع دليل استكشافي مستقل. التجارب أفكار تعليمية، ولا تمثل أماكن أو حجوزات فعلية.</p>{sources.map(s=><article className="source" key={s.url}><a href={s.url} target="_blank" rel="noreferrer">{s.title} ↗</a><p>{s.note}</p></article>)}<p>الرسوم أصلية ومؤقتة، وليست صورًا للأماكن. الخط: Noto Sans Arabic بترخيص SIL OFL؛ لم تتوفر ملفات مرخصة لخط «سعودي».</p></>:<><img src={image(item.image)} alt="رسم توضيحي مؤقت لطبيعة التجربة"/><span className="eyebrow">{item.status}</span><h2>{item.title}</h2><p className="location"><MapPin size={16}/>{item.city}</p><p>{item.description}</p><div className="notice"><Info size={18}/> هذه فكرة مقترحة، وليست حجزًا أو إعلانًا لمزرعة بعينها. تحقق من الموسم، والتوفر، وإذن الزيارة لدى الجهة المضيفة.</div></>}</dialog>}
export default function App(){const [selected,setSelected]=useState('SA.QS');const [tab,setTab]=useState('overview');const [city,setCity]=useState('all');const [modal,setModal]=useState(null);const info=useRef(null);const r=regions[selected];const choose=id=>{setSelected(id);setCity('all');setTab('overview');};const experiences=r?.experiences.filter(e=>city==='all'||e.city===city)||[];
 return <><header className="site-header"><a className="brand" href={import.meta.env.BASE_URL} aria-label="من هنا، الصفحة الرئيسية"><span className="brand-symbol"><Leaf size={26}/></span><span>من هنا<small>حكايات تنبت من الأرض</small></span></a><nav><span className="nav-active">الخريطة الزراعية</span><button onClick={()=>setModal({sources:true})}>عن الدليل <ArrowUpLeft size={14}/></button></nav><span className="header-tag"><span/> من أرضنا، إلى تجربتك</span></header>
 <main><section className="hero"><div className="eyebrow"><span className="tiny-line"/> المملكة العربية السعودية · دليل استكشافي</div><h1>لكل أرض <em>حكاية.</em></h1><div className="hero-row"><p>اكتشف محاصيل أرضنا، وعِش تجاربها</p><span>من واحات النخيل إلى مدرجات الجبال،<br/>رحلة تبدأ بنقطة على الخريطة.</span></div><div className="hero-ornament" aria-hidden="true"><Sprout size={155} strokeWidth={.7}/></div></section>
 <div className="workspace"><MapView selected={selected} onSelect={choose}/><section ref={info} className="info-panel" aria-label="معلومات المنطقة المختارة" key={selected}>
 {selected&&<button className="back-link" onClick={()=>choose(null)}><ArrowRight size={15}/> العودة للخريطة الرئيسية</button>}
 {!selected?<div className="empty-panel"><Compass size={48}/><h2>أين تبدأ حكايتك؟</h2><p>اختر إحدى المناطق الست على الخريطة لتكتشف محاصيلها وأفكار تجاربها الزراعية.</p></div>:!r?<div className="empty-panel"><Mountain size={48}/><span className="eyebrow">حكاية نكتبها قريبًا</span><h2>{regionNames[selected]}</h2><p>تظهر المنطقة بحدودها على الخريطة. محتواها الزراعي سيُضاف بعد التحقق من المصادر.</p><button className="primary" onClick={()=>choose('SA.QS')}>اكتشف القصيم <ChevronLeft size={16}/></button></div>:<><div className="region-heading"><div><span className="eyebrow">اكتشف المنطقة</span><h2>{r.name}</h2><p>{r.subtitle}</p></div><span className="region-number">{String(Object.keys(regions).indexOf(selected)+1).padStart(2,'0')}<small>/ 06</small></span></div><div className="region-image"><img src={image(r.image)} alt={`رسم توضيحي مؤقت: ${r.landscape}`} /><span><MapPin size={13}/>{r.landscape}</span><small>رسم توضيحي مؤقت</small></div><div className="tabs" role="tablist" aria-label="تفاصيل المنطقة">{[['overview','عن المنطقة'],['cities','المدن'],['experiences','التجارب']].map(([id,label],i)=><button key={id} id={`tab-${id}`} role="tab" aria-selected={tab===id} aria-controls="region-content" tabIndex={tab===id?0:-1} className={tab===id?'active':''} onClick={()=>setTab(id)} onKeyDown={e=>{if(['ArrowLeft','ArrowRight','Home','End'].includes(e.key)){e.preventDefault();const ids=['overview','cities','experiences'];const n=e.key==='Home'?0:e.key==='End'?2:(i+(e.key==='ArrowLeft'?1:2))%3;setTab(ids[n]);document.getElementById(`tab-${ids[n]}`).focus();}}}>{label}</button>)}</div><div id="region-content" role="tabpanel" aria-labelledby={`tab-${tab}`}>
 {tab==='overview'&&<><p className="description">{r.description}</p><div className="section-label"><span><Leaf size={16}/> من خيراتها</span><small>أبرز المحاصيل</small></div><div className="crop-cards">{r.crops.map((c,i)=><div className="crop-card" key={c}><img src={image(c.includes('بن')?'coffee':r.image)} alt="رسم توضيحي للمحصول"/><div><span className="crop-index">0{i+1}</span><strong>{c}</strong></div></div>)}</div></>}
 {(tab==='cities'||tab==='experiences')&&<p className="description">{tab==='cities'?'اختر مدينة لعرض أفكار التجارب المرتبطة بها. المدن هنا نطاقات للمحتوى المقترح، وليست عناوين لمزارع متاحة للزيارة.':'خطط لحكايتك القادمة: أفكار للتعرف على الزراعة ومحاصيل الأرض.'}</p>}
 <div className="section-label"><span><MapPin size={16}/> {tab==='cities'?'مدن تستحق الاكتشاف':'تجارب من الأرض'}</span><small>{experiences.length.toLocaleString('ar-SA')} أفكار</small></div><div className="city-filters" aria-label="تصفية حسب المدينة"><button onClick={()=>setCity('all')} aria-pressed={city==='all'} className={city==='all'?'active':''}><SlidersHorizontal size={13}/> كل المدن</button>{r.cities.map(c=><button key={c} onClick={()=>setCity(c)} aria-pressed={city===c} className={city===c?'active':''}>{c}</button>)}</div><div className="experience-list">{experiences.map(e=><button className="experience-card" key={e.id} onClick={()=>setModal(e)}><img src={image(e.image)} alt="رسم توضيحي مؤقت"/><span><small><MapPin size={11}/>{e.city} · تجربة مقترحة</small><strong>{e.title}</strong><span className="experience-description">{e.description}</span></span><span className="card-arrow"><ArrowUpLeft size={18}/></span></button>)}</div><p className="content-note"><Info size={13}/> محتوى زراعي أولي يحتاج توثيقًا قبل الاعتماد. التجارب أفكار وليست حجوزات.</p></div></>}
 </section></div><section className="journey-strip"><span className="journey-icon"><Leaf size={28}/></span><div><h3>تجربة أقرب للأرض</h3><p>تعرّف على المحصول، اسمع حكاية المزارع، واكتشف جمال التفاصيل.</p></div><span className="strip-note">اكتشاف · معرفة · أثر</span></section></main><footer><span>من هنا <small>© {new Date().getFullYear()} · خريطة السعودية الزراعية التفاعلية</small></span><button onClick={()=>setModal({sources:true})}>المصادر وحقوق الصور <ArrowUpLeft size={13}/></button><span>صُنع بروح الأرض <Leaf size={13}/></span></footer>{modal&&<Modal item={modal} onClose={()=>setModal(null)}/>}</>;
}
